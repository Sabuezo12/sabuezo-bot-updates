 internal static Icon ClientIcon(IntPtr window,IntPtr process) {

  if(window!=IntPtr.Zero&&IsWindow(window)) {

internal sealed class Client : IDisposable {
 internal readonly int Pid;internal readonly string Token;internal readonly IntPtr Handle;internal readonly long Created;
 internal string State="active",Label;internal long Seq=-1;internal int Minutes;internal uint LastActivity=Native.Tick,Prepared;
 internal IntPtr Window;internal uint Priority;internal Native.Power OriginalPower;internal bool CanPower,Powered,Hidden,WasMaximized;internal string EcoState="off";
 internal NotifyIcon Icon;private System.Drawing.Icon trayImage;private readonly Manager owner;private uint lastInput;
 internal Client(Manager owner,int pid,string token,string label) {
  this.owner=owner;Pid=pid;Token=token;Label=label;Handle=Native.Open(pid);Created=Native.Created(Handle);
  var input=new Native.Input();input.Size=8;Native.GetLastInputInfo(ref input);lastInput=input.Time;
 }
 internal string Base {get{return Path.Combine(owner.Root,Pid.ToString());}}
 internal bool Valid {get{return Native.Alive(Handle)&&Created!=0;}}
 internal void Status() {Native.Write(Base+".status","1\n"+Token+"\n"+Process.GetCurrentProcess().Id+"\n"+State+"\n"+Native.Now+"\n"+EcoState+"\n"+Window.ToInt64()+"\n");}
 internal void Prepare() {
  if(State!="active")return;Window=Native.Window(Pid,true);if(Window==IntPtr.Zero)return;
  WasMaximized=Native.Maximized(Window);Priority=Native.GetPriorityClass(Handle);
  if(Priority==0)return;
  CanPower=Native.ReadPower(Handle,out OriginalPower);
  Icon=new NotifyIcon();trayImage=Native.ClientIcon(Window,Handle);Icon.Icon=trayImage;
  string title=Native.ClientName(Handle)+": "+Label;
  Icon.Text=title.Substring(0,Math.Min(63,title.Length));
  // Wake disposes this icon and menu: run it after the click event returns.
  var menu=new ContextMenuStrip();menu.Items.Add("Restaurar "+Label,null,delegate { owner.Later(delegate { if(owner.Has(this))Wake(true,false); }); });
  menu.Items.Add("Desactivar modo AFK",null,delegate { owner.Later(delegate { if(owner.Has(this))Wake(true,true); }); });
  menu.Items.Add("Restaurar todos",null,delegate { owner.Later(owner.WakeAll); });
  Icon.ContextMenuStrip=menu;Icon.DoubleClick+=delegate { owner.Later(delegate { if(owner.Has(this))Wake(true,false); }); };Icon.Visible=true;
  // Persist recovery before any hiding or power changes. A replacement
  // companion can recover windows after an abrupt termination.
  Native.Write(Base+".recovery","1\n"+Pid+"\n"+Created+"\n"+Priority+"\n"+(WasMaximized?3:1)+"\n"+OriginalPower.ControlMask+"\n"+OriginalPower.StateMask+"\n");
  Prepared=Native.Tick;State="prepare";Status();
 }
 internal void Ready() {
  if(State!="prepare"||unchecked(Native.Tick-Prepared)>5000)return;
  Powered=CanPower&&Native.PowerBits(Handle,1,1);EcoState=Powered?"on":"unsupported";
  if(Powered)Native.SetPriorityClass(Handle,0x4000);
  if(Native.IsWindow(Window)){Native.ShowWindow(Window,0);Hidden=true;}
  State="afk";Status();
 }
 internal void Relax() {
  if(Powered){Native.PowerBits(Handle,OriginalPower.ControlMask,OriginalPower.StateMask);if(Native.GetPriorityClass(Handle)==0x4000)Native.SetPriorityClass(Handle,Priority);Powered=false;}
  EcoState="latency";Status();
 }
 internal void Wake(bool focus,bool pause) {
  if(Powered){Native.PowerBits(Handle,OriginalPower.ControlMask,OriginalPower.StateMask);if(Native.GetPriorityClass(Handle)==0x4000)Native.SetPriorityClass(Handle,Priority);Powered=false;}
  EcoState="off";
  if((Hidden||State=="prepare"||State=="afk")&&Window!=IntPtr.Zero&&Native.IsWindow(Window)){Native.ShowWindow(Window,Native.ShowCommand(Window,focus));if(focus)Native.SetForegroundWindow(Window);}
  Window=IntPtr.Zero;Hidden=false;ClearIcon();
  try{File.Delete(Base+".recovery");}catch{}
  LastActivity=Native.Tick;State=pause?"paused":"wake";Status();
 }
 internal void Command(string command,long seq,int minutes) {
  if(seq<=Seq)return;Seq=seq;Minutes=minutes;
  if(command=="release"){Wake(false,true);return;}
  if(command=="ready"){Ready();return;}
  if(command=="relax"){if(State=="afk")Relax();return;}
  if(command=="wake"){Wake(false,false);return;}
  if(command=="afk"){Prepare();return;}
  if(command=="auto"){if(State=="wake"||State=="paused"){State="active";LastActivity=Native.Tick;}Status();}
 }
 internal void Tick() {
  if(!Valid)return;
  uint now=Native.Tick;
  if(State=="prepare"&&unchecked(now-Prepared)>5000){Wake(false,false);return;}
  if(State=="afk") {
   // Windows may restore a window without granting it foreground focus.
   // Any visible AFK window must return to normal rendering and power policy.
   if(Native.IsWindowVisible(Window))Wake(false,false);
   return;
  }
  if(State!="active")return;
  var input=new Native.Input();input.Size=8;
  if(Native.GetLastInputInfo(ref input)&&input.Time!=lastInput) {
   lastInput=input.Time;uint pid;Native.GetWindowThreadProcessId(Native.GetForegroundWindow(),out pid);
   if(pid==Pid)LastActivity=now;
  }
  if(unchecked(now-LastActivity)>Minutes*60000U)Prepare();
 }
 private void ClearIcon(){if(Icon!=null){Icon.Visible=false;if(Icon.ContextMenuStrip!=null)Icon.ContextMenuStrip.Dispose();Icon.Dispose();Icon=null;}if(trayImage!=null){trayImage.Dispose();trayImage=null;}}
 public void Dispose(){try{if(Valid)Wake(false,true);}catch{}ClearIcon();if(Handle!=IntPtr.Zero)Native.CloseHandle(Handle);}
}

internal sealed class Manager : ApplicationContext {
 internal readonly string Root;private readonly Dictionary<int,Client> clients=new Dictionary<int,Client>();
 private readonly System.Windows.Forms.Timer timer=new System.Windows.Forms.Timer();private uint beat,empty;private int faults;private bool closing;
 internal Manager(string root){Root=root;Recover();timer.Interval=250;timer.Tick+=delegate{Tick();};timer.Start();Tick();}
 internal void WakeAll(){foreach(var c in new List<Client>(clients.Values))try{if(clients.ContainsKey(c.Pid))c.Wake(true,false);}catch{}}
 internal bool Has(Client c){Client x;return clients.TryGetValue(c.Pid,out x)&&x==c;}
 internal void Later(Action action){
  // Same UI thread, once the current click event has returned.
  var once=new System.Windows.Forms.Timer();once.Interval=1;
  once.Tick+=delegate { once.Stop();once.Dispose();try{if(!closing)action();}catch{} };
  once.Start();
 }
 private void Recover() {
  foreach(string path in Directory.GetFiles(Root,"*.recovery"))try {
   string[] p=File.ReadAllLines(path);int pid;long created;uint priority;int show;uint control=0,state=0;
   if(p.Length<5||p[0]!="1"||!Int32.TryParse(p[1],out pid)||!Int64.TryParse(p[2],out created)||!UInt32.TryParse(p[3],out priority)||!Int32.TryParse(p[4],out show))continue;
   if(p.Length>=7){UInt32.TryParse(p[5],out control);UInt32.TryParse(p[6],out state);}
   IntPtr h=Native.Open(pid);if(h!=IntPtr.Zero)try {
    if(Native.Alive(h)&&Native.Created(h)==created){Native.PowerBits(h,control,state);if(Native.GetPriorityClass(h)==0x4000)Native.SetPriorityClass(h,priority);IntPtr w=Native.Window(pid,false);if(w!=IntPtr.Zero)Native.ShowWindow(w,Native.ShowCommand(w,false));}
   }finally{Native.CloseHandle(h);}File.Delete(path);
  }catch{}
 }
 private void Tick() {
  if(closing)return;
  try {
   foreach(string path in Directory.GetFiles(Root,"*.control"))try {
    string[] p=File.ReadAllLines(path);int pid,minutes;long seq;
    if(p.Length!=7||p[0]!="1"||p[1].Length<8||p[1].Length>96||!Int32.TryParse(p[2],out pid)||pid<=0||
       Path.GetFileName(path)!=pid+".control"||!Int32.TryParse(p[3],out minutes)||minutes<1||minutes>240||!Int64.TryParse(p[5],out seq))continue;
    Client c;
    if(clients.TryGetValue(pid,out c)&&c.Token!=p[1]){c.Dispose();clients.Remove(pid);c=null;}
    if(c==null){if(p[4]=="release")continue;c=new Client(this,pid,p[1],p[6]);if(!c.Valid||File.GetLastWriteTimeUtc(path).ToFileTimeUtc()<c.Created){c.Dispose();continue;}clients.Add(pid,c);c.Status();}
    c.Command(p[4],seq,minutes);
    if(p[4]=="release"){c.Dispose();clients.Remove(pid);}
   }catch{}
   // One client's error (a locked file, a closed window) must not close the
   // companion for every other client.
   foreach(var c in new List<Client>(clients.Values))try{
    if(!c.Valid){c.Dispose();clients.Remove(c.Pid);try{File.Delete(c.Base+".control");}catch{}continue;}
    if(!File.Exists(c.Base+".control")){c.Dispose();clients.Remove(c.Pid);continue;}
    c.Tick();
   }catch{}
   uint now=Native.Tick;
   if(beat==0||unchecked(now-beat)>3000){try{Native.Write(Path.Combine(Root,"heartbeat"),Process.GetCurrentProcess().Id+"|"+Native.Now);beat=now;}catch{}}
   if(clients.Count==0){if(empty==0)empty=now;if(unchecked(now-empty)>15000)ExitThread();}else empty=0;
   faults=0;
  }catch{if(++faults>20)ExitThread();}
 }
 private void Cleanup(){if(closing)return;closing=true;timer.Stop();foreach(var c in clients.Values)c.Dispose();clients.Clear();try{File.Delete(Path.Combine(Root,"heartbeat"));}catch{}}
 protected override void ExitThreadCore(){Cleanup();base.ExitThreadCore();}
 protected override void Dispose(bool disposing){if(disposing){Cleanup();timer.Dispose();}base.Dispose(disposing);}
}
