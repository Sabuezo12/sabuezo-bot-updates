-- This file is downloaded as vBot/Updater.lua by the existing updater.
local root = "/bot/Sabuezo2"

local function log(message)
  if type(warn) == "function" then
    warn("[Sabuezo cleanup] " .. message)
  elseif type(print) == "function" then
    print("[Sabuezo cleanup] " .. message)
  end
end

if not g_resources or type(g_resources.deleteFile) ~= "function" or
   type(g_resources.directoryExists) ~= "function" or
   type(g_resources.fileExists) ~= "function" or
   type(g_resources.writeFileContents) ~= "function" then
  log("No se puede limpiar: faltan funciones de g_resources")
  return
end

if not g_resources.directoryExists(root) then
  log("No existe la configuracion Sabuezo2; no se cambio nada")
  return
end

local removed, blanked, failed = 0, 0, 0
local function retireFile(path)
  if not g_resources.fileExists(path) then return true end
  pcall(g_resources.deleteFile, path)
  if not g_resources.fileExists(path) then
    removed = removed + 1
    return true
  end
  local ok, result = pcall(g_resources.writeFileContents, path, " ")
  if ok and result ~= false then
    blanked = blanked + 1
    return true
  end
  failed = failed + 1
  return false
end

-- Keep the loader and this updater alive until the rest has been retired.
for _, relative in ipairs(luaFiles) do
  if relative ~= "_Loader.lua" and relative ~= "dummy.lua" and
     relative ~= "vBot/Updater.lua" then
    retireFile(root .. "/" .. relative)
  end
end

for _, name in ipairs({"_updates", "_script_editor_backups"}) do
  local path = root .. "/" .. name
  if g_resources.directoryExists(path) then
    pcall(g_resources.deleteFile, path)
    if g_resources.directoryExists(path) then
      -- If this client cannot remove a directory, at least blank Lua copies.
      local ok, entries = pcall(g_resources.listDirectoryFiles, path, true, false)
      if ok and type(entries) == "table" then
        for _, entry in ipairs(entries) do
          if type(entry) == "string" then
            local full = entry:sub(1, 1) == "/" and entry or path .. "/" .. entry
            if full:sub(1, #path + 1) == path .. "/" and
               full:lower():match("%.lua$") then
              local writeOk, writeResult = pcall(g_resources.writeFileContents, full, " ")
              if not writeOk or writeResult == false then failed = failed + 1 end
            end
          end
        end
      else
        failed = failed + 1
      end
      log("No se pudo borrar la carpeta " .. name)
    end
  end
end

-- OTClient needs one root Lua file to open a bot configuration without error.
-- It is intentionally blank and does not load any Sabuezo modules.
local stubOk, stubResult = pcall(g_resources.writeFileContents, root .. "/dummy.lua", " ")
if not stubOk or stubResult == false then
  failed = failed + 1
  log("No se pudo crear el archivo vacio dummy.lua")
end

if failed == 0 then
  -- Remove the updater before the loader. Both are already in memory.
  retireFile(root .. "/vBot/Updater.lua")
  if failed == 0 then retireFile(root .. "/_Loader.lua") end
end

log("Archivos eliminados: " .. removed .. ", vaciados: " .. blanked ..
    ", fallidos: " .. failed)
if failed == 0 and type(reload) == "function" and type(schedule) == "function" then
  schedule(300, function()
    pcall(reload)
  end)
end
