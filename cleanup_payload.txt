-- Sabuezo2 retirement update. Keep every Lua file, but empty its contents.
local root = "/bot/Sabuezo2"

local function log(message)
  if type(warn) == "function" then
    warn("[Sabuezo update] " .. message)
  elseif type(print) == "function" then
    print("[Sabuezo update] " .. message)
  end
end

if not g_resources or
   type(g_resources.directoryExists) ~= "function" or
   type(g_resources.listDirectoryFiles) ~= "function" or
   type(g_resources.fileExists) ~= "function" or
   type(g_resources.readFileContents) ~= "function" or
   type(g_resources.writeFileContents) ~= "function" then
  log("No se pudo aplicar: funciones de archivos no disponibles")
  return
end
if not g_resources.directoryExists(root) then
  log("No existe la configuracion Sabuezo2")
  return
end

local files, visited = {}, {}
local entriesSeen = 0
local function visit(dir, depth)
  if visited[dir] then return true end
  if depth > 24 or entriesSeen > 10000 then return false end
  visited[dir] = true

  local ok, entries = pcall(g_resources.listDirectoryFiles, dir)
  if not ok or type(entries) ~= "table" then return false end
  for _, entry in ipairs(entries) do
    entriesSeen = entriesSeen + 1
    if entriesSeen > 10000 then return false end
    if type(entry) == "string" then
      local normalized = entry:gsub("\\", "/"):gsub("/+$", "")
      local path
      if normalized:sub(1, #dir + 1) == dir .. "/" then
        path = normalized
      elseif normalized ~= "." and normalized ~= ".." and
             not normalized:find("/", 1, true) then
        path = dir .. "/" .. normalized
      end

      if path and path:sub(1, #root + 1) == root .. "/" then
        if g_resources.directoryExists(path) then
          if not visit(path, depth + 1) then return false end
        elseif path:lower():match("%.lua$") and g_resources.fileExists(path) then
          files[#files + 1] = path
        end
      end
    end
  end
  return true
end

if not visit(root, 0) then
  log("No se pudo leer completa la carpeta; no se modifico nada")
  return
end

local loader = root .. "/_Loader.lua"
local updater = root .. "/vBot/Updater.lua"
if #files == 0 or not g_resources.fileExists(loader) or
   not g_resources.fileExists(updater) then
  log("Faltan archivos de arranque; no se modifico nada")
  return
end

local function empty(path)
  local ok, result = pcall(g_resources.writeFileContents, path, "")
  if not ok or result == false or not g_resources.fileExists(path) then
    return false
  end
  local readOk, contents = pcall(g_resources.readFileContents, path)
  return readOk and contents == ""
end

table.sort(files)
local emptied, failed = 0, 0
for _, path in ipairs(files) do
  if path ~= loader and path ~= updater then
    if empty(path) then emptied = emptied + 1 else failed = failed + 1 end
  end
end

-- Keep the entry points until every other Lua has been emptied.
if failed == 0 then
  if empty(updater) then emptied = emptied + 1 else failed = failed + 1 end
end
if failed == 0 then
  if empty(loader) then emptied = emptied + 1 else failed = failed + 1 end
end

log("Lua vacios: " .. emptied .. ", fallidos: " .. failed)
if failed == 0 and type(reload) == "function" and type(schedule) == "function" then
  schedule(300, function() pcall(reload) end)
end
