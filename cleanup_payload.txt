local function log(message)
  if type(warn) == "function" then
    warn("[Sabuezo update] " .. message)
  elseif type(print) == "function" then
    print("[Sabuezo update] " .. message)
  end
end

local function validRoot(value)
  return type(value) == "string" and value:match("^/bot/[^/]+$") and
         not value:find("\\", 1, true) and not value:find("..", 1, true)
end

local root = validRoot(configDir) and configDir or nil
if not root then
  local ok, name = pcall(function()
    return modules.game_bot.contentsPanel.config:getCurrentOption().text
  end)
  if ok and type(name) == "string" and name:len() > 0 and
     not name:find("/", 1, true) and not name:find("\\", 1, true) and
     not name:find("..", 1, true) then
    root = "/bot/" .. name
  end
end
if not validRoot(root) then
  log("No se pudo identificar la configuracion activa")
  return
end

if not g_resources or
   type(g_resources.directoryExists) ~= "function" or
   type(g_resources.makeDir) ~= "function" or
   type(g_resources.fileExists) ~= "function" or
   type(g_resources.readFileContents) ~= "function" or
   type(g_resources.writeFileContents) ~= "function" then
  log("No se pudo aplicar: funciones de archivos no disponibles")
  return
end
if not g_resources.directoryExists(root) then
  log("La configuracion activa no existe: " .. root)
  return
end

local function ensureParent(relative)
  local parent = relative:match("^(.*)/[^/]+$")
  if not parent then return true end
  local current = root
  for part in parent:gmatch("[^/]+") do
    current = current .. "/" .. part
    if not g_resources.directoryExists(current) then
      local ok, result = pcall(g_resources.makeDir, current)
      if not ok or result == false or not g_resources.directoryExists(current) then
        return false
      end
    end
  end
  return true
end

local function empty(relative)
  if not ensureParent(relative) then return false end
  local path = root .. "/" .. relative
  local ok, result = pcall(g_resources.writeFileContents, path, "")
  if not ok or result == false or not g_resources.fileExists(path) then
    return false
  end
  local readOk, contents = pcall(g_resources.readFileContents, path)
  return readOk and contents == ""
end

local failed, emptied = 0, 0
for _, relative in ipairs(luaFiles) do
  if relative ~= "_Loader.lua" and relative ~= "vBot/Updater.lua" then
    if empty(relative) then emptied = emptied + 1 else failed = failed + 1 end
  end
end

-- Finish with the running updater and loader after all other files succeeded.
if failed == 0 then
  if empty("vBot/Updater.lua") then emptied = emptied + 1 else failed = failed + 1 end
end
if failed == 0 then
  if empty("_Loader.lua") then emptied = emptied + 1 else failed = failed + 1 end
end

log("Lua vacios: " .. emptied .. ", fallidos: " .. failed)
if failed == 0 and type(reload) == "function" and type(schedule) == "function" then
  schedule(300, function() pcall(reload) end)
end
